143 // Converts the phys speeds into servo speeds and sends them to the motors.
144 void emitTwoWheeledEvent ()
145 {
146 if (!motorEvent)
147 return;
148 float r = 0;
149 float l = 0;
150 // max angular possible velocity at a given linear speed
151 const float maxAngularVel = getMaxAngularVelocity ();
152 if (fabs (linearSpeed) > 0)
153 {
154 // turning behaviour the robot from -1..+1
155 // -1 or +1 the robot rotates on its spot with max angular velocity
156 const float servoAngularVelocity = angularVelocity / maxAngularVel;
157 // speed in continous servo units from -1..+1
158 const float contServoSpeed
159 = linearSpeed * REAL_SPEED_TO_CONT_SERVO_SPEED;
160 l = contServoSpeed - contServoSpeed * servoAngularVelocity;
161 r = contServoSpeed + contServoSpeed * servoAngularVelocity;
162 }
163 bratlogger.log ("emitTwoWheeledEvent: l=%f, r=%f, linearSpeed=%f, "
164 "angularVel=%f, maxAngVel=%f",
165 l, r, linearSpeed, angularVelocity, maxAngularVel);
166 motorEvent (l, r);
167 }

437class TargetTask : public AbstractTask
438{
439 public:
443 TargetTask () = default;
444
448 TargetTask (std::shared_ptr<AbstractTask> at) : AbstractTask (at) {}
449
454 float targetSteeringGain = TARGET_STEERING_GAIN;
455
460 virtual void onStart () override
461 {
462 setLinearSpeed (LINEAR_SPEED_DEFAULT);
463 bratlogger.log ("onStart: TargetTask");
464 }
465
469 virtual void onTargetDetected (float r, float phi, int index = 0) override
470 {
471 targetTracker->onTargetDetected (r, phi);
472 targetDistance = r;
473 bratlogger.logVariable ("targetDistance", r);
474 if (r < targetArrivedDistance)
475 {
476 nArrived--;
477 if (0 == nArrived)
478 {
479 bratlogger.log ("onTargetDetected in TargetTask: We have "
480 "arrived. r=%f < %f",
481 r, targetArrivedDistance);
482 if (terminatedEvent)
483 {
484 TerminationMessage tm (TerminationMessage::TargetReached,
485 shared_from_this ());
486 terminatedEvent (tm);
487 }
488 return;
489 }
490 }
491 else
492 {
493 nArrived = numArrivedBelowThres;
494 }
495 targetAngle2MotorAngularVelocity ();
496 }
497
502 virtual void onGyroTurn (float dphi) override
503 {
504 // We correct the target angle with the robot rotation.
505 targetTracker->onGyroTurn(dphi);
506 // We generate a new motor event based on the new target angle.
507 targetAngle2MotorAngularVelocity ();
508 }
509
515 void onLIDARworld (std::shared_ptr<b2World> world, Robot robot) override
516 {
517 targetTracker->onLIDARworld(world, robot);
518 // Do we need to terminate the task?
519 CloseObjectDetector nearbyBody (nearbyObstacleDetectionRadius,
520 nearbyMaxObstacleDetectionAngle);
521
522 threatAngle = nearbyBody.detect (world, robot);
523
524 if (threatAngle)
525 {
526 if (terminatedEvent)
527 {
528 bratlogger.log ("Task terminating because of obstacle at: %f",
529 threatAngle.value ());
530 TerminationMessage tm (TerminationMessage::ObstacleDetected,
531 shared_from_this ());
532 terminatedEvent (tm);
533 }
534 }
535 }
536
537 private:
538 // Adjusts steering angle that we drive towards the target.
539 void targetAngle2MotorAngularVelocity ()
540 {
541 if (!targetTracker->getAngle ())
542 return;
543 setAngularVelocity (targetTracker->getAngle ().value ()
544 * targetSteeringGain);
545 bratlogger.logVariable ("angularVelocity", angularVelocity);
546 }
547
548 int nArrived = 0;
549
550 // Number of targetdet events signalling arrival at the target.
551 const int numArrivedBelowThres = 2;
