171 void emitTwoWheeledEvent ()
172 {
173 if (!motorEvent)
174 return;
175 float r = 0;
176 float l = 0;
177 // max angular possible velocity at a given linear speed
178 const float maxAngularVel = getMaxAngularVelocity ();
179 if (fabs (linearSpeed) > 0)
180 {
181 // turning behaviour the robot from -1..+1
182 // -1 or +1 the robot rotates on its spot with max angular velocity
183 const float servoAngularVelocity = angularVelocity / maxAngularVel;
184 // speed in continous servo units from -1..+1
185 const float contServoSpeed
186 = linearSpeed * REAL_SPEED_TO_CONT_SERVO_SPEED;
187 l = contServoSpeed - contServoSpeed * servoAngularVelocity;
188 r = contServoSpeed + contServoSpeed * servoAngularVelocity;
189 }
190 bratlogger.log ("emitTwoWheeledEvent: l=%f, r=%f, linearSpeed=%f, "
191 "angularVel=%f, maxAngVel=%f",
192 l, r, linearSpeed, angularVelocity, maxAngularVel);
193 motorEvent (l, r);
194 }

536 TargetTask (std::shared_ptr<AbstractTask> at) : AbstractTask (at) {}

542 float targetSteeringGain = TARGET_STEERING_GAIN;

548 virtual void onStart () override

550 setLinearSpeed (LINEAR_SPEED_DEFAULT);
551 bratlogger.log ("onStart: TargetTask");
552 }
553
557 virtual void onTargetDetected (const float r, const float phi,
558 const int index = 0) override
559 {
560 targetTracker->onTargetDetected (phi);
561 targetDistance = r;
562 bratlogger.logVariable ("targetDistance", r);
563 logTargetAngle ();
564 if (r < targetArrivedDistance)
565 {
566 nArrived--;
567 if (0 == nArrived)
568 {
569 bratlogger.log ("onTargetDetected in TargetTask: We have "
570 "arrived. r=%f < %f",
571 r, targetArrivedDistance);
572 if (terminatedEvent)
573 {
574 TerminationMessage tm (TerminationMessage::TargetReached,
575 shared_from_this ());
576 terminatedEvent (tm);
577 }
578 return;
579 }
580 }
581 else
582 {
583 nArrived = numArrivedBelowThres;
584 }
585 targetAngle2MotorAngularVelocity ();
586 }
587
592 virtual void onGyroTurn (const float dphi) override
593 {
594 // We correct the target angle with the robot rotation.
595 targetTracker->onGyroTurn (dphi);
596 // loging how that was corrected
597 logTargetAngle ();
598 // We generate a new motor event based on the new target angle.
599 targetAngle2MotorAngularVelocity ();
600 }
601
607 void onLIDARworld (std::shared_ptr<b2World> world, Robot robot) override
608 {
609 targetTracker->onLIDARworld (world, robot);
610 // Do we need to terminate the task?
611 CloseObjectDetector nearbyBody (nearbyObstacleDetectionRadius,
612 nearbyMaxObstacleDetectionAngle);
613
614 threats = nearbyBody.detect (world, robot);
615
616 if (threats.hasThreats () > 0)
617 {
618 if (terminatedEvent)
619 {
620 bratlogger.log (
621 "Task terminating because of obstacle: l=%d, r=%d",
622 threats.nLeft, threats.nRight);
623 TerminationMessage tm (TerminationMessage::ObstacleDetected,
624 shared_from_this ());
625 terminatedEvent (tm);
626 }
627 }
628 }
629
630 private:
631 // Adjusts steering angle that we drive towards the target.
632 void targetAngle2MotorAngularVelocity ()
633 {
634 if (!targetTracker->getAngle ())
635 return;
636 setAngularVelocity (targetTracker->getAngle ().value ()
637 * targetSteeringGain);
638 bratlogger.logVariable ("angularVelocity", angularVelocity);
639 }
640

644 const int numArrivedBelowThres = 2;
