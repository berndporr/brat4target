108 if (!motorEvent)
109 return;
110 float r = 0;
111 float l = 0;
112 // max angular possible velocity at a given linear speed
113 const float maxAngularVel = getMaxAngularVelocity ();
114 if (fabs (linearSpeed) > 0)
115 {
116 // turning behaviour the robot from -1..+1
117 // -1 or +1 the robot rotates on its spot with max angular velocity
118 const float servoAngularVelocity = angularVelocity / maxAngularVel;
119 // speed in continous servo units from -1..+1
120 const float contServoSpeed
121 = linearSpeed * REAL_SPEED_TO_CONT_SERVO_SPEED;
122 l = contServoSpeed - contServoSpeed * servoAngularVelocity;
123 r = contServoSpeed + contServoSpeed * servoAngularVelocity;
124 }

126 "angularVel=%f, maxAngVel=%f",
