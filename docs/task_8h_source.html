102 void emitTwoWheeledEvent ()
103 {
104 if (!motorEvent)
105 return;
106 float r = 0;
107 float l = 0;
108 // max angular possible velocity at a given linear speed
109 const float maxAngularVel = getMaxAngularVelocity ();
110 if (fabs (linearSpeed) > 0)
111 {
112 // turning behaviour the robot from -1..+1
113 // -1 or +1 the robot rotates on its spot with max angular velocity
114 const float servoAngularVelocity = angularVelocity / maxAngularVel;
115 // speed in continous servo units from -1..+1
116 const float contServoSpeed
117 = linearSpeed * REAL_SPEED_TO_CONT_SERVO_SPEED;
118 l = contServoSpeed - contServoSpeed * servoAngularVelocity;
119 r = contServoSpeed + contServoSpeed * servoAngularVelocity;
120 }
121 bratlogger.log ("emitTwoWheeledEvent: l=%f, r=%f, linearSpeed=%f, "
122 "angularVel=%f, maxAngVel=%f",
123 l, r, linearSpeed, angularVelocity, maxAngularVel);
124 motorEvent (l, r);
125 }
