121 // Converts the phys speeds into servo speeds and sends them to the motors.
122 void emitTwoWheeledEvent ()
123 {
124 if (!motorEvent)
125 return;
126 float r = 0;
127 float l = 0;
128 // max angular possible velocity at a given linear speed
129 const float maxAngularVel = getMaxAngularVelocity ();
130 if (fabs (linearSpeed) > 0)
131 {
132 // turning behaviour the robot from -1..+1
133 // -1 or +1 the robot rotates on its spot with max angular velocity
134 const float servoAngularVelocity = angularVelocity / maxAngularVel;
135 // speed in continous servo units from -1..+1
136 const float contServoSpeed
137 = linearSpeed * REAL_SPEED_TO_CONT_SERVO_SPEED;
138 l = contServoSpeed - contServoSpeed * servoAngularVelocity;
139 r = contServoSpeed + contServoSpeed * servoAngularVelocity;
140 }
141 bratlogger.log ("emitTwoWheeledEvent: l=%f, r=%f, linearSpeed=%f, "
142 "angularVel=%f, maxAngVel=%f",
143 l, r, linearSpeed, angularVelocity, maxAngularVel);
144 motorEvent (l, r);
145 }
