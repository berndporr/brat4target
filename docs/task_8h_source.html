166 // Converts the phys speeds into servo speeds and sends them to the motors.
167 void emitTwoWheeledEvent ()
168 {
169 if (!motorEvent)
170 return;
171 float r = 0;
172 float l = 0;
173 // max angular possible velocity at a given linear speed
174 const float maxAngularVel = getMaxAngularVelocity ();
175 if (fabs (linearSpeed) > 0)
176 {
177 // turning behaviour the robot from -1..+1
178 // -1 or +1 the robot rotates on its spot with max angular velocity
179 const float servoAngularVelocity = angularVelocity / maxAngularVel;
180 // speed in continous servo units from -1..+1
181 const float contServoSpeed
182 = linearSpeed * REAL_SPEED_TO_CONT_SERVO_SPEED;
183 l = contServoSpeed - contServoSpeed * servoAngularVelocity;
184 r = contServoSpeed + contServoSpeed * servoAngularVelocity;
185 }
186 bratlogger.log ("emitTwoWheeledEvent: l=%f, r=%f, linearSpeed=%f, "
187 "angularVel=%f, maxAngVel=%f",
188 l, r, linearSpeed, angularVelocity, maxAngularVel);
189 motorEvent (l, r);
190 }

460class TargetTask : public AbstractTask
461{
462 public:
466 TargetTask () = default;
467
471 TargetTask (std::shared_ptr<AbstractTask> at) : AbstractTask (at) {}
472
477 float targetSteeringGain = TARGET_STEERING_GAIN;
478
483 virtual void onStart () override
484 {
485 setLinearSpeed (LINEAR_SPEED_DEFAULT);
486 bratlogger.log ("onStart: TargetTask");
487 }
488
492 virtual void onTargetDetected (const float r, const float phi, const int index = 0) override
493 {
494 targetTracker->onTargetDetected (phi);
495 targetDistance = r;
496 bratlogger.logVariable ("targetDistance", r);
497 if (r < targetArrivedDistance)
498 {
499 nArrived--;
500 if (0 == nArrived)
501 {
502 bratlogger.log ("onTargetDetected in TargetTask: We have "
503 "arrived. r=%f < %f",
504 r, targetArrivedDistance);
505 if (terminatedEvent)
506 {
507 TerminationMessage tm (TerminationMessage::TargetReached,
508 shared_from_this ());
509 terminatedEvent (tm);
510 }
511 return;
512 }
513 }
514 else
515 {
516 nArrived = numArrivedBelowThres;
517 }
518 targetAngle2MotorAngularVelocity ();
519 }
520
525 virtual void onGyroTurn (const float dphi) override
526 {
527 // We correct the target angle with the robot rotation.
528 targetTracker->onGyroTurn(dphi);
529 // We generate a new motor event based on the new target angle.
530 targetAngle2MotorAngularVelocity ();
531 }
532
538 void onLIDARworld (std::shared_ptr<b2World> world, Robot robot) override
539 {
540 targetTracker->onLIDARworld(world, robot);
541 // Do we need to terminate the task?
542 CloseObjectDetector nearbyBody (nearbyObstacleDetectionRadius,
543 nearbyMaxObstacleDetectionAngle);
544
545 threatAngle = nearbyBody.detect (world, robot);
546
547 if (threatAngle)
548 {
549 if (terminatedEvent)
550 {
551 bratlogger.log ("Task terminating because of obstacle at: %f",
552 threatAngle.value ());
553 TerminationMessage tm (TerminationMessage::ObstacleDetected,
554 shared_from_this ());
555 terminatedEvent (tm);
556 }
557 }
558 }
559
560 private:
561 // Adjusts steering angle that we drive towards the target.
562 void targetAngle2MotorAngularVelocity ()
563 {
564 if (!targetTracker->getAngle ())
565 return;
566 setAngularVelocity (targetTracker->getAngle ().value ()
567 * targetSteeringGain);
568 bratlogger.logVariable ("angularVelocity", angularVelocity);
569 }
570
571 int nArrived = 0;
572
573 // Number of targetdet events signalling arrival at the target.
574 const int numArrivedBelowThres = 2;
