200 terminationMessage2task (AbstractTask::TerminationMessage tm)
201 {
202 bratlogger.log ("Reactive: onTaskTerminated");
203 switch (tm.terminationReason)
204 {
205 case AbstractTask::TerminationMessage::TargetDetected:
206 bratlogger.log (
207 "ReactiveConfigurator: Target detected. We are targeting.");
208 return std::make_unique<TargetTask> (tm.taskWhichSentMessage);
209 case AbstractTask::TerminationMessage::ObstacleDetected:
210 bratlogger.log ("ReactiveConfigurator: Obstacle detected.");
211 if (tm.taskWhichSentMessage->getThreatAngle ())
212 {
213 if (tm.taskWhichSentMessage->getThreatAngle ().value () > 0)
214 {
215 bratlogger.log ("ReactiveConfigurator: Avoidance to the "
216 "RIGHT b/c obstacle to the left.");
217 return std::make_unique<AvoidTask> (
218 tm.taskWhichSentMessage,
219 AvoidTask::AvoidDirecton::RIGHT);
220 }
221 else
222 {
223 bratlogger.log ("ReactiveConfigurator: Avoidance to the "
224 "LEFT b/c obstacle to the right.");
225 return std::make_unique<AvoidTask> (
226 tm.taskWhichSentMessage,
227 AvoidTask::AvoidDirecton::LEFT);
228 }
229 }
230 else
231 {
232 if (random () & 1)
233 {
234 bratlogger.log ("ReactiveConfigurator: Avoidance to the "
235 "LEFT b/c random choice.");
236 return std::make_unique<AvoidTask> (
237 tm.taskWhichSentMessage,
238 AvoidTask::AvoidDirecton::LEFT);
239 }
240 else
241 {
242 bratlogger.log ("ReactiveConfigurator: Avoidance to the "
243 "RIGHT b/c random choice.");
244 return std::make_unique<AvoidTask> (
245 tm.taskWhichSentMessage,
246 AvoidTask::AvoidDirecton::LEFT);
247 }
248 }
249 break;
250 case AbstractTask::TerminationMessage::ObstacleAvoided:
251 bratlogger.log ("ReactiveConfigurator: Obstacle avoided. Now "
252 "targeting again.");
253 return std::make_unique<TargetTask> (tm.taskWhichSentMessage);
254 case AbstractTask::TerminationMessage::TargetReached:
255 bratlogger.log ("ReactiveConfigurator: TargetReached: stopping.");
256 return std::make_unique<EndTask> (tm.taskWhichSentMessage);
257 case AbstractTask::TerminationMessage::INVALID:
258 bratlogger.log (
259 "ReactiveConfigurator: Message==INVALID (BUG?): stopping.");
260 return std::make_unique<EndTask> (tm.taskWhichSentMessage);
261 }
262 bratlogger.log (
263 "ReactiveConfigurator: Message==??? (BUG?): stopping.");
264 return std::make_unique<EndTask> (tm.taskWhichSentMessage);
265 }
