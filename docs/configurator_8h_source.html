169 // static so that the tests can test it without touching the configurator state.
170 static std::shared_ptr<AbstractTask>
171 terminationMessage2task (AbstractTask::TerminationMessage tm)
172 {
173 bratlogger.log ("Reactive: onTaskTerminated");
174 switch (tm.terminationReason)
175 {
176 case AbstractTask::TerminationMessage::TargetDetected:
177 bratlogger.log (
178 "ReactiveConfigurator: Target detected. We are targeting.\n");
179 return std::make_shared<TargetTask> (tm.taskWhichSentMessage);
180 case AbstractTask::TerminationMessage::ObstacleDetected:
181 bratlogger.log ("ReactiveConfigurator: Obstacle detected.\n");
182 if (tm.taskWhichSentMessage->getThreatAngle ())
183 {
184 if (tm.taskWhichSentMessage->getThreatAngle ().value () > 0)
185 {
186 bratlogger.log ("ReactiveConfigurator: Avoidance to the "
187 "RIGHT b/c obstacle to the left.\n");
188 return std::make_shared<AvoidTask> (
189 tm.taskWhichSentMessage,
190 AvoidTask::AvoidDirecton::RIGHT);
191 }
192 else
193 {
194 bratlogger.log ("ReactiveConfigurator: Avoidance to the "
195 "LEFT b/c obstacle to the right.\n");
196 return std::make_shared<AvoidTask> (
197 tm.taskWhichSentMessage,
198 AvoidTask::AvoidDirecton::LEFT);
199 }
200 }
201 else
202 {
203 if (random () & 1)
204 {
205 bratlogger.log ("ReactiveConfigurator: Avoidance to the "
206 "LEFT b/c random choice.\n");
207 return std::make_shared<AvoidTask> (
208 tm.taskWhichSentMessage,
209 AvoidTask::AvoidDirecton::LEFT);
210 }
211 else
212 {
213 bratlogger.log ("ReactiveConfigurator: Avoidance to the "
214 "RIGHT b/c random choice.\n");
215 return std::make_shared<AvoidTask> (
216 tm.taskWhichSentMessage,
217 AvoidTask::AvoidDirecton::LEFT);
218 }
219 }
220 break;
221 case AbstractTask::TerminationMessage::ObstacleAvoided:
222 bratlogger.log ("ReactiveConfigurator: Obstacle avoided. Now "
223 "targeting again.\n");
224 return std::make_shared<TargetTask> (tm.taskWhichSentMessage);
225 case AbstractTask::TerminationMessage::TargetReached:
226 bratlogger.log (
227 "ReactiveConfigurator: TargetReached: stopping.\n");
228 return std::make_shared<StopTask> (tm.taskWhichSentMessage);
229 case AbstractTask::TerminationMessage::INVALID:
230 bratlogger.log (
231 "ReactiveConfigurator: Message==INVALID (BUG?): stopping.\n");
232 return std::make_shared<StopTask> (tm.taskWhichSentMessage);
233 }
234 bratlogger.log (
235 "ReactiveConfigurator: Message==??? (BUG?): stopping.\n");
236 return std::make_shared<StopTask> (tm.taskWhichSentMessage);
237 }
