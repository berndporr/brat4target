184 terminationMessage2task (AbstractTask::TerminationMessage tm)
185 {
186 bratlogger.log ("Reactive: onTaskTerminated");
187 switch (tm.terminationReason)
188 {
189 case AbstractTask::TerminationMessage::TargetDetected:
190 bratlogger.log (
191 "ReactiveConfigurator: Target detected. We are targeting.");
192 return std::make_shared<TargetTask> (tm.taskWhichSentMessage);
193 case AbstractTask::TerminationMessage::ObstacleDetected:
194 bratlogger.log ("ReactiveConfigurator: Obstacle detected.");
195 if (tm.taskWhichSentMessage->getThreatAngle ())
196 {
197 if (tm.taskWhichSentMessage->getThreatAngle ().value () > 0)
198 {
199 bratlogger.log ("ReactiveConfigurator: Avoidance to the "
200 "RIGHT b/c obstacle to the left.");
201 return std::make_shared<AvoidTask> (
202 tm.taskWhichSentMessage,
203 AvoidTask::AvoidDirecton::RIGHT);
204 }
205 else
206 {
207 bratlogger.log ("ReactiveConfigurator: Avoidance to the "
208 "LEFT b/c obstacle to the right.");
209 return std::make_shared<AvoidTask> (
210 tm.taskWhichSentMessage,
211 AvoidTask::AvoidDirecton::LEFT);
212 }
213 }
214 else
215 {
216 if (random () & 1)
217 {
218 bratlogger.log ("ReactiveConfigurator: Avoidance to the "
219 "LEFT b/c random choice.");
220 return std::make_shared<AvoidTask> (
221 tm.taskWhichSentMessage,
222 AvoidTask::AvoidDirecton::LEFT);
223 }
224 else
225 {
226 bratlogger.log ("ReactiveConfigurator: Avoidance to the "
227 "RIGHT b/c random choice.");
228 return std::make_shared<AvoidTask> (
229 tm.taskWhichSentMessage,
230 AvoidTask::AvoidDirecton::LEFT);
231 }
232 }
233 break;
234 case AbstractTask::TerminationMessage::ObstacleAvoided:
235 bratlogger.log ("ReactiveConfigurator: Obstacle avoided. Now "
236 "targeting again.");
237 return std::make_shared<TargetTask> (tm.taskWhichSentMessage);
238 case AbstractTask::TerminationMessage::TargetReached:
239 bratlogger.log ("ReactiveConfigurator: TargetReached: stopping.");
240 return std::make_shared<StopTask> (tm.taskWhichSentMessage);
241 case AbstractTask::TerminationMessage::INVALID:
242 bratlogger.log (
243 "ReactiveConfigurator: Message==INVALID (BUG?): stopping.");
244 return std::make_shared<StopTask> (tm.taskWhichSentMessage);
245 }
246 bratlogger.log (
247 "ReactiveConfigurator: Message==??? (BUG?): stopping.");
248 return std::make_shared<StopTask> (tm.taskWhichSentMessage);
249 }
